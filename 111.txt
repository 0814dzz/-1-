import tkinter as tk
from tkinter import scrolledtext, ttk
import re
import nltk

# --- NLTK setup (safe if already downloaded) ---
try:
    nltk.data.find('tokenizers/punkt')
except LookupError:
    nltk.download('punkt')

try:
    nltk.data.find('taggers/averaged_perceptron_tagger')
except LookupError:
    nltk.download('averaged_perceptron_tagger')


# ---------- Simple NLP helper functions ----------

def tokenize_text(text: str):
    """Tokenize text into words using NLTK (fallback to regex)."""
    text = text.strip()
    if not text:
        return []
    try:
        return nltk.word_tokenize(text)
    except LookupError:
        return re.findall(r"\w+|\S", text)


def pos_tag(tokens):
    """POS tag tokens using NLTK (fallback: mark all as 'NN')."""
    if not tokens:
        return []
    try:
        return nltk.pos_tag(tokens)
    except LookupError:
        return [(t, "NN") for t in tokens]


def word_frequencies(tokens):
    """Very small frequency counter."""
    freqs = {}
    for t in tokens:
        lower = t.lower()
        freqs[lower] = freqs.get(lower, 0) + 1
    # sort by frequency descending
    return sorted(freqs.items(), key=lambda x: (-x[1], x[0]))


# ---------- Rule-based chatbot core ----------

class NLPChatbot:
    

    def __init__(self):
        # very small knowledge base
        self.intents = [
            ("hello", "Hi! 👋 I am an NLP teaching bot. Ask me about tokens, POS tags, or NLP."),
            ("today is happy.", "i don't have a specific answer for that yet.but I can tell you that your message has 4 tokens and 4 unique words. Click 'Show NLP Analysis' to see more details!"),  
]

    def generate_response(self, user_text: str) -> str:
        text = user_text.lower().strip()

        # 1. Simple intent matching
        for key, answer in self.intents:
            if key in text:
                return answer

        # 2. Simple echo with NLP flavour
        tokens = tokenize_text(user_text)
        if not tokens:
            return "Please type something so I can analyse it 🙂."

        num_tokens = len(tokens)
        unique_words = len(set(t.lower() for t in tokens))

        return (
            f"I don't have a specific answer for that yet.\n"
            f"But I can tell you that your message has {num_tokens} tokens "
            f"and {unique_words} unique words. "
            f"Click 'Show NLP Analysis' to see more details!"
        )


# ---------- GUI Application ----------

class NLPChatGUI:
    def __init__(self, root):
        self.root = root
        self.root.title("NLP Teaching Chatbot")

        # nice minimum size
        self.root.geometry("900x550")

        self.bot = NLPChatbot()

        # --- Main layout: left = chat, right = analysis ---
        self.main_frame = ttk.Frame(root, padding=10)
        self.main_frame.pack(fill=tk.BOTH, expand=True)

        self.left_frame = ttk.Frame(self.main_frame)
        self.left_frame.pack(side=tk.LEFT, fill=tk.BOTH, expand=True)

        self.right_frame = ttk.Frame(self.main_frame)
        self.right_frame.pack(side=tk.RIGHT, fill=tk.BOTH, expand=False)

        # ---------- Chat area ----------

        title_label = ttk.Label(
            self.left_frame,
            text="NLP Chatbot",
            font=("Times New Roman", 16, "bold")
        )
        title_label.pack(anchor="w", pady=(0, 5))

        self.chat_box = scrolledtext.ScrolledText(
            self.left_frame,
            wrap=tk.WORD,
            state=tk.DISABLED,
            font=("Times New Roman", 11)
        )
        self.chat_box.pack(fill=tk.BOTH, expand=True)

        # user input
        input_frame = ttk.Frame(self.left_frame)
        input_frame.pack(fill=tk.X, pady=(5, 0))

        self.user_entry = ttk.Entry(input_frame, font=("Times New Roman", 11))
        self.user_entry.pack(side=tk.LEFT, fill=tk.X, expand=True)
        self.user_entry.bind("<Return>", self.on_send)

        send_button = ttk.Button(input_frame, text="Send", command=self.on_send)
        send_button.pack(side=tk.LEFT, padx=(5, 0))

        # ---------- Analysis panel ----------

        analysis_label = ttk.Label(
            self.right_frame,
            text="NLP Analysis",
            font=("Times New Roman", 14, "bold")
        )
        analysis_label.pack(anchor="w")

        self.show_analysis_var = tk.BooleanVar(value=True)
        show_chk = ttk.Checkbutton(
            self.right_frame,
            text="Show NLP Analysis",
            variable=self.show_analysis_var
        )
        show_chk.pack(anchor="w", pady=(0, 5))

        # tokens
        ttk.Label(self.right_frame, text="Tokens:", font=("Times New Roman", 11, "bold")).pack(anchor="w")
        self.tokens_box = scrolledtext.ScrolledText(
            self.right_frame,
            height=8,
            width=35,
            wrap=tk.WORD,
            state=tk.DISABLED,
            font=("Times New Roman", 10)
        )
        self.tokens_box.pack(fill=tk.X, pady=(0, 5))

        # POS tags
        ttk.Label(self.right_frame, text="POS Tags:", font=("Times New Roman", 11, "bold")).pack(anchor="w")
        self.pos_box = scrolledtext.ScrolledText(
            self.right_frame,
            height=8,
            width=35,
            wrap=tk.WORD,
            state=tk.DISABLED,
            font=("Times New Roman", 10)
        )
        self.pos_box.pack(fill=tk.X, pady=(0, 5))

        # word frequencies
        ttk.Label(self.right_frame, text="Word Frequencies:", font=("Times New Roman", 11, "bold")).pack(anchor="w")
        self.freq_box = scrolledtext.ScrolledText(
            self.right_frame,
            height=6,
            width=35,
            wrap=tk.WORD,
            state=tk.DISABLED,
            font=("Times New Roman", 10)
        )
        self.freq_box.pack(fill=tk.X)

        # greeting message
        self.add_chat_message("Bot", "Hello! I am your NLP teaching assistant.\n"
                                      "Type a message and press Enter or 'Send'.")

    # ---------- Chat + Analysis logic ----------

    def add_chat_message(self, speaker: str, text: str):
        self.chat_box.config(state=tk.NORMAL)
        self.chat_box.insert(tk.END, f"{speaker}: ", ("speaker",))
        self.chat_box.insert(tk.END, text + "\n\n")
        self.chat_box.tag_config("speaker", font=("Times New Roman", 11, "bold"))
        self.chat_box.config(state=tk.DISABLED)
        self.chat_box.yview(tk.END)

    def clear_analysis(self):
        for box in (self.tokens_box, self.pos_box, self.freq_box):
            box.config(state=tk.NORMAL)
            box.delete("1.0", tk.END)
            box.config(state=tk.DISABLED)

    def update_analysis(self, user_text: str):
        if not self.show_analysis_var.get():
            self.clear_analysis()
            return

        tokens = tokenize_text(user_text)
        tags = pos_tag(tokens)
        freqs = word_frequencies(tokens)

        # tokens
        self.tokens_box.config(state=tk.NORMAL)
        self.tokens_box.delete("1.0", tk.END)
        self.tokens_box.insert(tk.END, " | ".join(tokens))
        self.tokens_box.config(state=tk.DISABLED)

        # pos tags
        self.pos_box.config(state=tk.NORMAL)
        self.pos_box.delete("1.0", tk.END)
        for word, tag in tags:
            self.pos_box.insert(tk.END, f"{word:<15} {tag}\n")
        self.pos_box.config(state=tk.DISABLED)

        # frequencies
        self.freq_box.config(state=tk.NORMAL)
        self.freq_box.delete("1.0", tk.END)
        for word, count in freqs:
            self.freq_box.insert(tk.END, f"{word:<15} {count}\n")
        self.freq_box.config(state=tk.DISABLED)

    def on_send(self, event=None):
        user_text = self.user_entry.get().strip()
        if not user_text:
            return
        self.user_entry.delete(0, tk.END)

        # show user message
        self.add_chat_message("You", user_text)

        # NLP analysis panel
        self.update_analysis(user_text)

        # bot response
        bot_reply = self.bot.generate_response(user_text)
        self.add_chat_message("Bot", bot_reply)


# ---------- Run the app ----------

if __name__ == "__main__":
    root = tk.Tk()
    app = NLPChatGUI(root)
    root.mainloop()
